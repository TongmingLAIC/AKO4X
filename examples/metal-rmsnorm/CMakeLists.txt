cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS version")
project(ako_metal_rmsnorm LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "The native example requires macOS and Metal")
endif()
add_executable(rmsnorm_bench main.mm)
target_compile_features(rmsnorm_bench PRIVATE cxx_std_20)
set_target_properties(rmsnorm_bench PROPERTIES OBJCXX_STANDARD 20)
target_compile_options(rmsnorm_bench PRIVATE -fobjc-arc -Wall -Wextra)
target_compile_definitions(rmsnorm_bench PRIVATE AKO_KERNEL_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/rmsnorm.metal")
target_link_libraries(rmsnorm_bench PRIVATE "-framework Foundation" "-framework Metal")
